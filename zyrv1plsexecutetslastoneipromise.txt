-- ═══════════════════════════════════════════════════════
-- DELAYED NAMECALL HOOK (starts at 30s)
-- ═══════════════════════════════════════════════════════

task.delay(30, function()
    local ok = pcall(function()
        local old = nil

        old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            local method = getnamecallmethod()

            if method == "FireServer" or method == "InvokeServer" then
                local args = {}
                local count = select("#", ...)

                for i = 1, count do
                    local arg = select(i, ...)
                    if type(arg) == "string" or type(arg) == "number" then
                        args[i] = tostring(arg)
                    elseif type(arg) == "boolean" then
                        args[i] = tostring(arg)
                    else
                        args[i] = type(arg)
                    end
                end

                table.insert(remotes, {
                    method = method,
                    remote = self.Name,
                    path = self:GetFullName(),
                    args = args
                })
            end

            return old(self, ...)
        end)
    end)

    if ok then
        print("[ACTIVE] Surveillance running. Press P to dump.")
    else
        print("[WARN] Namecall hook failed on this executor.")
    end
end)

-- ═══════════════════════════════════════════════════════
-- MEMORY EXTRACTION (starts at 45s)
-- ═══════════════════════════════════════════════════════

task.delay(45, function()
    print("[EXTRACT] Scanning loaded functions...")

    -- Check available tools
    local canGetInfo = false
    local canGetUpvalue = false
    local canGetConnections = false

    pcall(function()
        if debug.getinfo then canGetInfo = true end
    end)
    pcall(function()
        if debug.getupvalue then canGetUpvalue = true end
    end)
    pcall(function()
        if getconnections then canGetConnections = true end
    end)

    print("[EXTRACT] getinfo: " .. tostring(canGetInfo))
    print("[EXTRACT] getupvalue: " .. tostring(canGetUpvalue))
    print("[EXTRACT] getconnections: " .. tostring(canGetConnections))

    -- Walk RunService heartbeat callbacks
    if canGetConnections then
        local ok, connections = pcall(getconnections, RunService.Heartbeat)

        if ok and connections then
            print("[EXTRACT] Heartbeat connections: " .. #connections)

            for i, conn in ipairs(connections) do
                if conn.Function then
                    local func = conn.Function
                    local info = nil

                    if canGetInfo then
                        pcall(function()
                            info = debug.getinfo(func)
                        end)
                    end

                    table.insert(closures, {
                        signal = "Heartbeat",
                        index = i,
                        source = info and info.source or "?",
                        nups = info and info.nups or 0
                    })

                    if canGetUpvalue then
                        for idx = 1, 100 do
                            local ok2, name, value = pcall(
                                debug.getupvalue, func, idx
                            )
                            if not ok2 or not name then break end

                            local val = "?"
                            if type(value) == "string" then
                                val = value:sub(1, 100)
                            elseif type(value) == "number" then
                                val = tostring(value)
                            elseif type(value) == "boolean" then
                                val = tostring(value)
                            end

                            table.insert(upvalues, {
                                signal = "Heartbeat",
                                index = i,
                                upIdx = idx,
                                name = name,
                                value = val,
                                vtype = type(value)
                            })
                        end
                    end
                end
            end
        end
    end

    print("[EXTRACT] Done. " .. #closures .. " closures, "
        .. #upvalues .. " upvalues found.")
    print("[DUMP] Press P to dump everything.")
end)

-- ═══════════════════════════════════════════════════════
-- DUMP (press P)
-- ═══════════════════════════════════════════════════════

local dumpText = nil
local chunkNum = 0
local CHUNK = 3000

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.P then

        if not dumpText then
            -- Build dump
            local out = "=== RESISTANCE DUMP ===\n"
            out = out .. "Remotes: " .. #remotes .. "\n"
            out = out .. "Closures: " .. #closures .. "\n"
            out = out .. "Upvalues: " .. #upvalues .. "\n\n"

            -- Remotes
            out = out .. "--- REMOTES ---\n"
            for i, r in ipairs(remotes) do
                out = out .. i .. ". " .. r.method .. " -> " .. r.path .. "\n"
                for j, a in ipairs(r.args) do
                    out = out .. "   arg" .. j .. ": " .. a .. "\n"
                end
            end

            -- Closures
            out = out .. "\n--- CLOSURES ---\n"
            for i, c in ipairs(closures) do
                out = out .. i .. ". " .. (c.signal or "?")
                    .. " src=" .. (c.source or "?")
                    .. " nups=" .. tostring(c.nups) .. "\n"
            end

            -- Upvalues
            out = out .. "\n--- UPVALUES ---\n"
            for i, u in ipairs(upvalues) do
                out = out .. i .. ". " .. (u.name or "?")
                    .. " (" .. (u.vtype or "?") .. ") = "
                    .. (u.value or "?") .. "\n"
            end

            dumpText = out
            chunkNum = 0
            print("[DUMP] " .. #out .. " chars ready")
        end

        -- Copy chunk to clipboard
        chunkNum = chunkNum + 1
        local start = (chunkNum - 1) * CHUNK + 1
        local stop = math.min(chunkNum * CHUNK, #dumpText)

        if start > #dumpText then
            print("[DUMP] Complete!")
            dumpText = nil
            chunkNum = 0
            return
        end

        setclipboard("CHUNK " .. chunkNum .. "/"
            .. math.ceil(#dumpText / CHUNK) .. "\n"
            .. dumpText:sub(start, stop))
        print("[DUMP] Chunk " .. chunkNum .. " on clipboard")
    end
end)
