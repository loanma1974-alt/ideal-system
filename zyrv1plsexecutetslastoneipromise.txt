local function extractFunctions()

    local hasDebugGetinfo = false
    local hasDebugGetupvalue = false
    local hasGetconnections = false

    pcall(function()
        if debug.getinfo then hasDebugGetinfo = true end
    end)
    pcall(function()
        if debug.getupvalue then hasDebugGetupvalue = true end
    end)

        if getconnections then hasGetconnections = true end
    end)

    print("[PHASE2] debug.getupvalue: " .. tostring(hasDebugGetupvalue))
    print("[PHASE2] getconnections: " .. tostring(hasGetconnections))

    -- Method 1: Walk UI for button callbacks
    print("[PHASE2] Walking UI tree...")

    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then
        pcall(function()
            if gethui then
                playerGui = gethui()
            end
        end)
    end

    if playerGui then
        local uiCount = 0

        local function walkGui(parent, depth)
            if depth > 20 then return end
            for _, child in ipairs(parent:GetChildren()) do
                uiCount = uiCount + 1

                if hasGetconnections then
                    if child:IsA("TextButton") or child:IsA("GuiButton")
                       or child:IsA("ImageButton") then

                        local ok, connections = pcall(getconnections, child.MouseButton1Click)
                        if ok and connections then
                            for i, conn in ipairs(connections) do
                                if conn.Function then
                                    local func = conn.Function
                                    local info = nil
                                    if hasDebugGetinfo then
                                        pcall(function()
                                            info = debug.getinfo(func)
                                        end)
                                    end

                                    log("closures", {
                                        type = "button_callback",
                                        element = child.Name,
                                        source = info and info.source or "unknown",
                                        what = info and info.what or "unknown",
                                        linedefined = info and info.linedefined or 0,
                                        nups = info and info.nups or 0
                                    })

                                    -- Dump upvalues (this gives us variable names
                                    -- and values from the closure's captured scope)
                                    if hasDebugGetupvalue then
                                        for upIndex = 1, 100 do
                                            local ok2, uvName, uvValue = pcall(
                                                debug.getupvalue, func, upIndex

                                            if type(uvValue) == "string"
                                               or type(uvValue) == "number"
                                               or type(uvValue) == "boolean" then
                                                valueStr = tostring(uvValue)
                                            elseif type(uvValue) == "Instance" then

                                            end

                                            log("upvalues", {
                                                element = child.Name,
                                                upIndex = upIndex,
                                                name = uvName,
                                                value = valueStr,
                                                type = type(uvValue)
                                            })
                                        end
                                    end
                                end
                            end
                        end
                    end
                end

                walkGui(child, depth + 1)
            end
        end

        walkGui(playerGui, 0)
        print("[PHASE2] Walked " .. uiCount .. " UI elements")
    end

    -- Method 2: Extract RunService callback info
    print("[PHASE2] Extracting RunService callbacks...")

    if hasGetconnections then
        local signals = {
            { RunService.Heartbeat, "Heartbeat" },
            { RunService.RenderStepped, "RenderStepped" },
            { RunService.Stepped, "Stepped" },
        }

        for _, signalInfo in ipairs(signals) do
            local signal = signalInfo[1]
            local signalName = signalInfo[2]

            local ok, connections = pcall(getconnections, signal)
            if ok and connections then
                print("[PHASE2] " .. signalName .. ": " .. #connections .. " connections")

                for i, conn in ipairs(connections) do
                    if conn.Function then
                        local func = conn.Function
                        local info = nil
                        if hasDebugGetinfo then
                            pcall(function()
                                info = debug.getinfo(func)
                            end)
                        end

                        log("closures", {
                            type = "runservice_callback",
                            signal = signalName,
                            index = i,
                            source = info and info.source or "unknown",
                            what = info and info.what or "unknown",
                            linedefined = info and info.linedefined or 0,
                            nups = info and info.nups or 0

            end

    print("[PHASE2] Scanning globals...")

    for key, value in pairs(genv) do
        if type(value) == "function" then
            local info = nil
            if hasDebugGetinfo then
                pcall(function()
                    info = debug.getinfo(value)
                end)
            end

            if info and info.source and info.source ~= "=[C]" then

                    type = "global_function",
                    name = key,

                        end

                            global = key,
                            upIndex = upIndex,
                            name = uvName,
                            value = valueStr,
                            type = type(uvValue)

                    end

    end

    print("[PHASE2] Closures: " .. #captured.closures)
    print("[PHASE2] Upvalues: " .. #captured.upvalues)

local fullDump = nil
local currentChunk = 0
local CHUNK_SIZE = 3000

local function buildDump()
    local out = "=== LURAPH v3.1 EXTRACTION ===\n"
    out = out .. "Remotes: " .. #captured.remotes .. "\n"
    out = out .. "Closures: " .. #captured.closures .. "\n"
    out = out .. "Upvalues: " .. #captured.upvalues .. "\n\n"

    out = out .. "=== REMOTE CALLS ===\n"
    for i, entry in ipairs(captured.remotes) do
        local d = entry.data
        if d.method then
            out = out .. "[" .. i .. "] " .. d.method .. " -> " .. d.path .. "\n"
            for j, arg in ipairs(d.args or {}) do
                out = out .. "  arg[" .. j .. "] = " .. arg .. "\n"
            end
        end
    end

    out = out .. "\n=== CLOSURES ===\n"
    for i, entry in ipairs(captured.closures) do
        local d = entry.data
        out = out .. "[" .. i .. "] " .. (d.type or "?")
        if d.element then out = out .. " element=" .. d.element end
        if d.signal then out = out .. " signal=" .. d.signal end
        if d.name then out = out .. " name=" .. d.name end
        if d.source then out = out .. " source=" .. d.source end
        if d.what then out = out .. " what=" .. d.what end
        out = out .. "\n"
    end

    out = out .. "\n=== UPVALUES ===\n"
    for i, entry in ipairs(captured.upvalues) do
        local d = entry.data
        out = out .. "[" .. i .. "] " .. (d.name or "?")
            .. " (" .. (d.type or "?") .. ") = "
            .. (d.value or "complex") .. "\n"
    end

    out = out .. "\n=== BYPASS LOG ===\n"
    for i, entry in ipairs(captured.bypassLog) do
        out = out .. "[" .. i .. "] " .. (entry.url or "?") .. "\n"
    end

    return out
end

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end

    if input.KeyCode == Enum.KeyCode.P then
        if not fullDump then
            fullDump = buildDump()
            local total = math.ceil(#fullDump / CHUNK_SIZE)
            print("[DUMP] " .. #fullDump .. " chars, " .. total .. " chunks")
            currentChunk = 0
        end

        currentChunk = currentChunk + 1
        local startPos = (currentChunk - 1) * CHUNK_SIZE + 1
        local endPos = math.min(currentChunk * CHUNK_SIZE, #fullDump)

        if startPos > #fullDump then
            print("[DUMP] All chunks copied!")
            fullDump = nil
            currentChunk = 0
            return
        end

        setclipboard("CHUNK " .. currentChunk .. "/"
            .. math.ceil(#fullDump / CHUNK_SIZE) .. "\n"
            .. fullDump:sub(startPos, endPos))
        print("[DUMP] Chunk " .. currentChunk .. " copied — paste, press P for next")
    end

    if input.KeyCode == Enum.KeyCode.O then
        extractFunctions()
    end
end)
