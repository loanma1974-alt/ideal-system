-- OBSERVER v2 — reviewed, bug-fixed
-- Load this, then load resistance. Surveillance at 30s.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local remotes = {}
local upvalues = {}
local closures = {}
local oldNamecall = nil

print("[OK] Load the resistance script now.")

-- ═════════════════════════════════════════════
-- NAMECALL HOOK (defined now, activated at 30s)
-- ═════════════════════════════════════════════

local function namecallHook(self, ...)
    -- Always pass through first — if logging fails,
    -- the game still works normally
    if oldNamecall then
        local method = getnamecallmethod()

        if method == "FireServer" or method == "InvokeServer" then
            -- Logging wrapped in pcall — errors here
            -- can never break the namecall chain
            pcall(function()
                local args = {}
                local count = select("#", ...)

                for i = 1, count do
                    local arg = select(i, ...)
                    local argType = type(arg)

                    if argType == "string" or argType == "number" or argType == "boolean" then
                        args[i] = tostring(arg)
                    else
                        args[i] = argType
                    end
                end

                table.insert(remotes, {
                    method = method,
                    remote = self.Name,
                    path = self:GetFullName(),
                    args = args
                })
            end)
        end

        return oldNamecall(self, ...)
    end
end

task.delay(30, function()
    local ok = pcall(function()
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(namecallHook))
    end)

    if ok then
        print("[ACTIVE] Surveillance running. Press P to dump.")
    else
        print("[WARN] Hook failed on this executor.")
    end
end)

-- ═════════════════════════════════════════════
-- MEMORY EXTRACTION (activates at 45s)
-- ═════════════════════════════════════════════

local function extractMemory()
    print("[EXTRACT] Scanning...")

    local canGetInfo = false
    local canGetUpvalue = false
    local canGetConnections = false

    pcall(function()
        if debug.getinfo then canGetInfo = true end
    end)
    pcall(function()
        if debug.getupvalue then canGetUpvalue = true end
    end)
    pcall(function()
        if getconnections then canGetConnections = true end
    end)

    print("[EXTRACT] getinfo=" .. tostring(canGetInfo)
        .. " getupvalue=" .. tostring(canGetUpvalue)
        .. " getconnections=" .. tostring(canGetConnections))

    if not canGetConnections then
        print("[EXTRACT] getconnections unavailable. Skipping.")
        return
    end

    local ok, connections = pcall(getconnections, RunService.Heartbeat)

    if not ok or not connections then
        print("[EXTRACT] Failed to get connections.")
        return
    end

    print("[EXTRACT] Heartbeat connections: " .. #connections)

    for i, conn in ipairs(connections) do
        if conn.Function then
            local func = conn.Function

            -- Get function metadata
            if canGetInfo then
                pcall(function()
                    local info = debug.getinfo(func)
                    table.insert(closures, {
                        signal = "Heartbeat",
                        index = i,
                        source = info.source or "?",
                        what = info.what or "?",
                        nups = info.nups or 0
                    })
                end)
            else
                table.insert(closures, {
                    signal = "Heartbeat",
                    index = i,
                    source = "unavailable"
                })
            end

            -- Extract upvalues (captured variables inside closures)
            if canGetUpvalue then
                for idx = 1, 100 do
                    local gotUp, name, value = pcall(debug.getupvalue, func, idx)

                    if not gotUp or not name then
                        break
                    end

                    local val = "?"

                    if type(value) == "string" then
                        val = value:sub(1, 100)
                    elseif type(value) == "number" then
                        val = tostring(value)
                    elseif type(value) == "boolean" then
                        val = tostring(value)
                    end

                    table.insert(upvalues, {
                        signal = "Heartbeat",
                        index = i,
                        upIdx = idx,
                        name = name,
                        value = val,
                        vtype = type(value)
                    })
                end
            end
        end
    end

    print("[EXTRACT] Done. " .. #closures .. " closures, " .. #upvalues .. " upvalues.")
    print("[DUMP] Press P to dump.")
end

task.delay(45, extractMemory)

-- ═════════════════════════════════════════════
-- DUMP (press P for clipboard chunks)
-- ═════════════════════════════════════════════

local dumpText = nil
local chunkNum = 0
local CHUNK = 3000

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then
        return
    end

    if input.KeyCode == Enum.KeyCode.P then
        if not dumpText then
            local out = "=== RESISTANCE DUMP ===\n"
            out = out .. "Remotes: " .. #remotes .. "\n"
            out = out .. "Closures: " .. #closures .. "\n"
            out = out .. "Upvalues: " .. #upvalues .. "\n\n"

            out = out .. "--- REMOTES ---\n"
            for i, r in ipairs(remotes) do
                out = out .. i .. ". " .. r.method .. " -> " .. r.path .. "\n"
                for j, a in ipairs(r.args) do
                    out = out .. "   arg" .. j .. ": " .. a .. "\n"
                end
            end

            out = out .. "\n--- CLOSURES ---\n"
            for i, c in ipairs(closures) do
                out = out .. i .. ". " .. (c.signal or "?")
                    .. " src=" .. (c.source or "?")
                    .. " nups=" .. tostring(c.nups or 0) .. "\n"
            end

            out = out .. "\n--- UPVALUES ---\n"
            for i, u in ipairs(upvalues) do
                out = out .. i .. ". " .. (u.name or "?")
                    .. " (" .. (u.vtype or "?") .. ") = "
                    .. (u.value or "?") .. "\n"
            end

            dumpText = out
            chunkNum = 0
            print("[DUMP] " .. #out .. " chars ready")
        end

        chunkNum = chunkNum + 1

        local start = (chunkNum - 1) * CHUNK + 1
        local stop = math.min(chunkNum * CHUNK, #dumpText)

        if start > #dumpText then
            print("[DUMP] Complete!")
            dumpText = nil
            chunkNum = 0
            return
        end

        setclipboard("CHUNK " .. chunkNum .. "/"
            .. math.ceil(#dumpText / CHUNK) .. "\n"
            .. dumpText:sub(start, stop))
        print("[DUMP] Chunk " .. chunkNum .. " on clipboard")
    end
end)
