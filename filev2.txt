local realLoadstring = loadstring

-- Hook the GLOBAL loadstring
-- Luraph's wrapper grabs `loadstring` at the top — it reads
-- from the global environment. If we replace _G.loadstring
-- BEFORE the wrapper executes, the wrapper stores OUR hooked
-- version in variable `r`.

_G.loadstring = function(str, chunkname)
    if type(str) == "string" and #str > 50 then
        local isBinary = str:sub(1, 1) ~= nil and str:byte(1) == 0x1B

        print(string.format("[SOURCE] loadstring called: %d chars, %s, name=%s",
            #str,
            isBinary and "BYTECODE" or "TEXT/LUA",
            chunkname or "unnamed"
        ))

        log("chunks", {
            type = isBinary and "bytecode" or "text",
            name = chunkname or "unnamed",
            size = #str,
            source = str, -- THE ACTUAL DECODED PROGRAM
            binary = isBinary
        })
    end
    return realLoadstring(str, chunkname)
end

-- ═══════════════════════════════════════════════════════════════

local fullDump = nil
local currentChunk = 0
local CHUNK_SIZE = 3000

local function buildDump()
    local out = "=== LURAPH SOURCE EXTRACTION ===\n"
    out = out .. "Chunks captured: " .. #captured.chunks .. "\n"
    out = out .. "Remotes captured: " .. #captured.remotes .. "\n\n"

    -- THE SOURCE CHUNKS
    out = out .. "=== DECODED PROGRAMS ===\n"
    for i, chunk in ipairs(captured.chunks) do
        local d = chunk.data
        out = out .. "\n--- CHUNK " .. i .. " ---\n"
        out = out .. "Type: " .. d.type .. "\n"
        out = out .. "Name: " .. d.name .. "\n"
        out = out .. "Size: " .. d.size .. " chars\n"
        if d.binary then
            out = out .. "(BINARY BYTECODE — needs decompiler)\n"
        else
            out = out .. "SOURCE:\n"
            out = out .. d.source .. "\n"
        end
    end

    -- REMOTES
    out = out .. "\n=== REMOTE CALLS ===\n"
    for i, entry in ipairs(captured.remotes) do
        local d = entry.data
        out = out .. "[" .. i .. "] " .. d.method .. " -> " .. d.path .. "\n"
        for j, arg in ipairs(d.args) do
            out = out .. "  arg[" .. j .. "] = " .. arg .. "\n"
        end
    end

    return out
end

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.P then
        if not fullDump then
            fullDump = buildDump()
            local total = math.ceil(#fullDump / CHUNK_SIZE)
            print("[DUMP] " .. #fullDump .. " chars, " .. total .. " chunks ready")
            currentChunk = 0
        end

        currentChunk = currentChunk + 1
        local start = (currentChunk - 1) * CHUNK_SIZE + 1
        local stop = math.min(currentChunk * CHUNK_SIZE, #fullDump)

        if start > #fullDump then
            print("[DUMP] All chunks copied!")
            fullDump = nil
            currentChunk = 0
            return
        end

        setclipboard("CHUNK " .. currentChunk .. "/" ..
            math.ceil(#fullDump / CHUNK_SIZE) .. "\n" .. fullDump:sub(start, stop))
        print("[DUMP] Chunk " .. currentChunk .. " copied — paste it, press P for next")
    end
end)
